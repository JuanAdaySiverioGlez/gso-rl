% GCM table row: \filagcm{y}{color}{name}{cost}
\newcommand{\filagcm}[4]{%
  \fill[#2] (0.15,#1) circle (3pt);
  \node[anchor=west, font=\scriptsize\bfseries] at (0.35,#1) {#3};
  \node[anchor=east, font=\scriptsize] at (2.05,#1) {#4};
}
% Agent row: \filaagente{y}{color}{macro-move}{bar length}{visits}
\newcommand{\filaagente}[5]{%
  \node[anchor=west, font=\scriptsize\bfseries, #2] at (0,#1) {#3};
  \fill[#2!70] (0.7,#1-0.11) rectangle ++(#4,0.22);
  \node[anchor=west, font=\tiny, gray!70!black] at (2.75,#1) {$n=#5$};
}
% CCM cell: \casilla[box style]{x}{colour}{macro-move}{cost}
% (macro-move and cost are separate nodes: see the dvisvgm note in common.tex)
\newcommand{\casilla}[5][]{%
  \node[draw=gray!70!black, fill=white, rounded corners=2pt, minimum width=0.95cm,
        minimum height=0.45cm, #1] at (#2,0.5) {};
  \node[#3, font=\scriptsize\bfseries, anchor=east, inner sep=0pt] at (#2-0.06,0.5) {#4};
  \node[font=\scriptsize, anchor=west, inner sep=0pt] at (#2-0.06,0.5) {#5};
}
