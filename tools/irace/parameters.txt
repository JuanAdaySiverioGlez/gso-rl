## GSO-RL parameters for irace (docs/07-tuning.md). One line per parameter:
##   name   switch   type   range   | condition
## Types: c categorical, i integer, r real, o ordered. The switch is what target_runner.py receives;
## "--x@arg=" is the argument `arg` of the component in parameter x (folded into x's text spec).
## Comment out what you do not want to tune; add your adapter's problem.* parameters at the end.
## Ranges are deliberately wide; narrow them when you know your problem.

## Learning agent (docs/04)
agent_type      "--agent.type="                       c (thompson, ucb1, epsilon_greedy, softmax, discounted_ucb, sliding_window_ucb)
ucb1_c          "--agent.type@c="                     r,log (0.01, 2.0)       | agent_type == "ucb1"
eps             "--agent.type@eps="                   r (0.01, 0.4)           | agent_type == "epsilon_greedy"
tau             "--agent.type@tau="                   r,log (0.005, 1.0)      | agent_type == "softmax"
ducb_c          "--agent.type@c="                     r,log (0.01, 2.0)       | agent_type == "discounted_ucb"
ducb_gamma      "--agent.type@gamma="                 r (0.9, 0.999)          | agent_type == "discounted_ucb"
swucb_c         "--agent.type@c="                     r,log (0.01, 2.0)       | agent_type == "sliding_window_ucb"
swucb_window    "--agent.type@window="                i (20, 1000)            | agent_type == "sliding_window_ucb"
on_replace      "--agent.on_replace="                 c (reset, decay)
decay_lambda    "--agent.on_replace@lambda="          r (0.1, 0.9)            | on_replace == "decay"

## Reward: Eq. 3 of the article (alpha weighs the improvement of the solution, 1 - alpha the distance
## to the best). Needs "--reward.type=eq3" in the "fixed" list of runner.json.
reward_alpha    "--reward.type@alpha="                r (0.0, 1.0)

## Memory and regions (docs/02)
gcm_size        "--gcm.size="                         i (5, 30)
tenure          "--gcm.tenure="                       r (0.05, 1.0)
case_a          "--gcm.case_a="                       c (drift, renew)
delta_ratio     "--region.delta_ratio="               r (0.1, 0.9)
rings           "--region.rings="                     i (1, 3)
on_saturation   "--region.on_saturation="             c (none, replace)
saturation      "--region.saturation="                i,log (3, 100)          | on_saturation == "replace"

## Search loop
region_iters    "--search.region_iters="              i (3, 30)
guide_sel       "--search.guide_selection="           c (ucb, geometric, uniform)
guide_ucb_c     "--search.guide_selection@c="         r (0.1, 3.0)            | guide_sel == "ucb"
guide_geo_p     "--search.guide_selection@p="         r (0.05, 0.6)           | guide_sel == "geometric"
agent_of        "--search.agent_of="                  c (start, closest)
ccm_size        "--ccm.size="                         i (0, 10)
